# ProjectZeo Policy Configuration Template
# ==========================================
# GAP-5 FIX: This file was completely absent from the repository.
# PolicyEngine loads policy.yaml at task start. Without this file,
# operators had no way to know:
#   - The file exists and controls system behavior
#   - What keys are supported
#   - What the defaults are
#   - How to customize the policy for their environment
#
# USAGE:
#   1. Copy this file: cp policy.yaml.template policy.yaml
#   2. Edit policy.yaml to match your environment
#   3. Do NOT commit policy.yaml with real paths to git (add to .gitignore)
#   4. ProjectZeo reads policy.yaml at each task start (hot-reload safe)
#
# All sections are optional — omit any section to use the built-in defaults.
# ==========================================

  - "scp"
  - "sftp"

# ─────────────────────────────────────────────────────────────────────────────
# FILESYSTEM POLICY
# Controls where the agent is allowed to create/modify files.
# ─────────────────────────────────────────────────────────────────────────────
filesystem:
  # Directories the agent MAY write to (prefix match)
  allowed_write_paths:
    - "/tmp/"
    - "/home/"
    - "~/Desktop/"
    - "~/Documents/"
    - "~/Downloads/"
    - "~/projects/"
    - "/var/tmp/"

  # Directories the agent is NEVER allowed to write to (hard deny, prefix match)
  denied_write_paths:
    - "/etc/"
    - "/sys/"
    - "/proc/"
    - "/dev/"
    - "/boot/"
    - "/root/"
    - "/usr/bin/"
    - "/usr/sbin/"
    - "/bin/"
    - "/sbin/"

# ─────────────────────────────────────────────────────────────────────────────
# NETWORK POLICY
# Controls what network operations the agent may perform.
# ─────────────────────────────────────────────────────────────────────────────
network:
  # Allow outbound HTTP/HTTPS (required for install, research, web tasks)
  allow_outbound_http: true

  # Allow outbound SSH (set false unless the agent explicitly needs remote access)
  allow_outbound_ssh: false

  # Denied domains — agent cannot resolve or connect to these via browser/curl
  denied_domains:
    - "ngrok.io"
    - "pagekite.net"

# ─────────────────────────────────────────────────────────────────────────────
# COMMAND EXECUTION POLICY
# ─────────────────────────────────────────────────────────────────────────────
commands:
  # Require human confirmation for any command that uses sudo
  require_confirmation_for_sudo: true

  # Maximum time in seconds for a single shell command (0 = use system default)
  max_command_timeout_seconds: 300

  # Shell to use for command execution (leave empty for system default)
  shell: ""

# ─────────────────────────────────────────────────────────────────────────────
# BROWSER AUTOMATION POLICY (requires playwright — GAP-1 fix)
# ─────────────────────────────────────────────────────────────────────────────
browser:
  # Use Playwright DOM automation for browser-targeted actions when available.
  # Falls back to pyautogui coordinate clicks if playwright is not installed.
  prefer_playwright: true

  # Browser to launch for Playwright sessions
  # Options: "chromium", "firefox", "webkit"
  playwright_browser: "chromium"

  # Launch browser in headless mode (false = visible browser window)
  headless: false

  # Wait for network idle before considering a navigation complete (ms)
  navigation_timeout_ms: 30000

# ─────────────────────────────────────────────────────────────────────────────
# RESTORATION POLICY
# ─────────────────────────────────────────────────────────────────────────────
restoration:
  # Whether to attempt snapshot/restore after each task
  enabled: true

  # Restoration verification tolerance for cursor position (pixels)
  cursor_tolerance_px: 5

  # Timeout for the restoration phase (seconds)
  timeout_seconds: 60

# ─────────────────────────────────────────────────────────────────────────────
# LOGGING AND AUDIT
# ─────────────────────────────────────────────────────────────────────────────
logging:
  # Journal log level: DEBUG | INFO | WARNING | ERROR
  level: "INFO"

  # Maximum journal log file size in MB before rotation
  max_size_mb: 50

  # Keep last N journal files
  keep_files: 5
